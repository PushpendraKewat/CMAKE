
cmake_minimum_required(VERSION 3.20)

project(CustomCommand LANGUAGES CXX)

add_custom_command(
    OUTPUT generated.txt
    COMMAND ${CMAKE_COMMAND} -E echo "Generated by CMake!" > generated.txt
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    COMMENT "Generating text file..."
)

add_custom_target(GenerateFile ALL
    DEPENDS generated.txt
)
